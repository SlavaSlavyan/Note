```cpp
#include <iostream>
#include <string>

class User {
private:
    std::string name;
    int age;

public:
    // Базовый (целевой) конструктор
    User(std::string userName, int userAge) 
        : name(userName), age(userAge) 
    {
        std::cout << "Основной конструктор вызван. Имя: " << name << ", Возраст: " << age << "\n";
    }

    // Делегирующий конструктор (принимает только имя)
    User(std::string userName) 
        : User(userName, 0) // Делегирование первому конструктору
    {
        std::cout << "Делегирующий конструктор (только имя)\n";
    }

    // Делегирующий конструктор по умолчанию
    User() 
        : User("Неизвестно") // Делегирование второму конструктору
    {
        std::cout << "Делегирующий конструктор по умолчанию\n";
    }
};

int main() {
    User user1; // Запустит цепочку вызовов: User() -> User("Неизвестно") -> User("Неизвестно", 0)
    return 0;
}
```
